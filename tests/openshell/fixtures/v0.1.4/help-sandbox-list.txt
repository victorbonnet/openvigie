List sandboxes

USAGE
  openshell sandbox list [OPTIONS]

FLAGS:
      --page-size <PAGE_SIZE>
          Maximum number of sandboxes to return in this page
          
          [default: 100]

      --page-token <PAGE_TOKEN>
          Opaque continuation token from a previous page
          
          [default: ""]

      --ids
          Print only sandbox ids (one per line)

      --names
          Print only sandbox names (one per line)

      --selector <SELECTOR>
          Filter sandboxes by label selector (key1=value1,key2=value2)

  -o, --output <OUTPUT>
          Output format
          
          [default: table]
          [possible values: table, yaml, json]

      --all-workspaces
          List sandboxes across all workspaces (overrides --workspace)

GATEWAY FLAGS:
  -g, --gateway <GATEWAY>
          Gateway name to operate on (resolved from stored metadata)
          
          [env: OPENSHELL_GATEWAY=]

      --gateway-endpoint <GATEWAY_ENDPOINT>
          Gateway endpoint URL (e.g. <https://gateway.example.com>). Connects directly without looking up gateway metadata
          
          [env: OPENSHELL_GATEWAY_ENDPOINT=]

      --gateway-insecure
          Skip TLS certificate verification for gateway connections
          
          [env: OPENSHELL_GATEWAY_INSECURE=]

GLOBAL FLAGS:
      --workspace <WORKSPACE>
          Workspace scope for resource operations
          
          [env: OPENSHELL_WORKSPACE=]
          [default: default]

  -v, --verbose...
          Increase verbosity (-v, -vv, -vvv)

      --color <WHEN>
          When to colorize output. `auto` colorizes only when stdout is a terminal

          Possible values:
          - auto:   Colorize a stream only when that stream is a terminal and no environment override applies
          - always: Always colorize, even when output is redirected
          - never:  Never colorize
          
          [env: OPENSHELL_COLOR=]
          [default: auto]

  -h, --help
          Print help

  -V, --version
          Print version
